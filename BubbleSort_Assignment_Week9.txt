
#include <iostream>
#include <vector>
using namespace std;

void bubbleSortDescending(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; ++i) {
        for (int j = 0; j < n - i - 1; ++j) {
            if (arr[j] < arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
            }
        }
    }
}

bool isAlmostSorted(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n; ++i) {
        if (i > 0 && arr[i] < arr[i - 1]) {
            if (arr[i] < arr[i - 1] - 1) return false;
        }
        if (i < n - 1 && arr[i] > arr[i + 1]) {
            if (arr[i] > arr[i + 1] + 1) return false;
        }
    }
    return true;
}

int main() {
    vector<int> arr = {1, 6, 2, 5, 4, 3};
    bubbleSortDescending(arr);
    for (int num : arr) {
        cout << num << " ";
    }
    cout << endl;

    vector<int> checkArr = {1, 2, 3, 5, 4, 6};
    if (isAlmostSorted(checkArr)) {
        cout << "The array is almost sorted." << endl;
    } else {
        cout << "The array is not almost sorted." << endl;
    }

    return 0;
}
